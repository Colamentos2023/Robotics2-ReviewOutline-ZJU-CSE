\subsection{动力学模型}
\begin{enumerate}
    \item \textbf{坐标系变换与旋转矩阵基础}\footnote{考虑到本部分同学们已经在《机器人学I》课程中学习过，此处仅做简要整理}
    \[
    \mathbf{v}^3 =
    \underbrace{%
    \begin{bmatrix}
    1 & 0 & 0\\[4pt]
    0 & \cos\phi & \sin\phi\\[4pt]
    0 & -\sin\phi & \cos\phi
    \end{bmatrix}
    \begin{bmatrix}
    \cos\theta & 0 & -\sin\theta\\[4pt]
    0 & 1 & 0\\[4pt]
    \sin\theta & 0 & \cos\theta
    \end{bmatrix}
    \begin{bmatrix}
    \cos\psi & \sin\psi & 0\\[4pt]
    -\sin\psi & \cos\psi & 0\\[4pt]
    0 & 0 & 1
    \end{bmatrix}%
    }_{\mathbf{R}_0^3}
    \;
    \underbrace{\begin{bmatrix}\alpha\\[4pt]\beta\\[4pt]\gamma\end{bmatrix}}_{\mathbf{v}^0}
    \]
    
    \begin{itemize}
      \item 不同坐标系可通过三次旋转达到重合。
      \item 三次旋转的顺序可以不同，但机器人领域中默认旋转顺序为 Z–Y–X。
      \item Z–Y–X 旋转顺序对应的角称为欧拉角，记为 $\psi,\;\theta,\;\phi$。
      \item 欧拉角确定以后，旋转矩阵 $\mathbf{R}$ 也唯一确定。
      \item 旋转矩阵确定后，向量在不同坐标系下的坐标可以通过该旋转矩阵进行变换。
      \item 对换性质：$\mathbf{R}_1^2=\mathbf{R}_2^{1^T}=\mathbf{R}_2^{1^{-1}}$。
    \end{itemize}

    \item \textbf{四旋翼无人机的坐标系、状态定义}

        \item 螺旋桨转速 $\omega_i$\footnote{由转速能确定三轴的力矩，来改变无人机的位姿}

\subsection{飞行控制}
\begin{enumerate}
    \item \textbf{解耦}

$$
\begin{bmatrix}\dot{\phi}\\[4pt]\dot{\theta}\\[4pt]\dot{\psi}\end{bmatrix}
=
\begin{bmatrix}
1 & \tan\theta\sin\phi & \tan\theta\cos\phi\\[6pt]
0 & \cos\phi & -\sin\phi\\[6pt]
0 & \dfrac{\sin\phi}{\cos\theta} & \dfrac{\cos\phi}{\cos\theta}
\end{bmatrix}
\begin{bmatrix}p\\[4pt]q\\[4pt]r\end{bmatrix}
$$

$$
\begin{bmatrix}\dot{p}\\[4pt]\dot{q}\\[4pt]\dot{r}\end{bmatrix}
=
\big(\mathbf{I}^B\big)^{-1}
\left(
\begin{bmatrix}L\\[4pt]M\\[4pt]N\end{bmatrix}^B
-
\begin{bmatrix}p\\[4pt]q\\[4pt]r\end{bmatrix}
\times
\mathbf{I}^B
\begin{bmatrix}p\\[4pt]q\\[4pt]r\end{bmatrix}
\right)
$$
不难发现，这仍然是一个多输入多输出问题，角速度耦合在一起，为了变成SISO系统，我们强行将其他角速度置零，即认为无人机只进行Roll、Yaw、Pitch中的一个，例如：
$$
\phi=0,\quad \psi=0,\quad p=0,\quad r=0,\quad L=0,\quad N=0
$$
最终得到：
$$
\begin{cases}
\dot{\theta}=q,\\[6pt]
\dot{q}=\dfrac{M}{I_{yy}},
\end{cases}
$$
同理也可以得到：
$$
\begin{cases}
\dot{\phi}=p,\\[6pt]
\dot{p}=\dfrac{L}{I_{xx}},
\end{cases}
\qquad
\begin{cases}
\dot{\psi}=r,\\[6pt]
\dot{r}=\dfrac{N}{I_{zz}}.
\end{cases}
$$
这样就形成了独立、单入单出（SISO）的线性系统，方便进一步控制。
\item \textbf{四旋翼俯仰角速率控制器}\\

以无人机只进行俯仰为例，即：
$$
\begin{cases}
\dot{\theta}=q,\\[6pt]
\dot{q}=\dfrac{M}{I_{yy}},
\end{cases}
$$
在此基础上加一个反馈，构建如下图所示的控制律：
\begin{figure}[H]
    \centering
    \includegraphics[width=0.6\linewidth]{images/52.png}
    \caption{俯仰速率控制}
\end{figure}
因为$I_{yy}$是一个很小的量，因此
$$\frac{q(s)}{q_{r}(s)}=\frac{K_{p}}{I_{yy}s+K_{p}}\approx1$$
意味着我们的$q$可以紧紧跟上$q_r$，几乎是要多少角速率有多少角速率；其中$K_p$越大，响应越快。

\item \textbf{四旋翼俯仰姿态控制器}\\
在上一小节的基础上，如果希望对角度$\theta$也进行控制，我们只需要加一个积分器：

\begin{figure}[H]

其中：
\begin{itemize}
    \item 外环控制器计算需要多少\textbf{角速率}来跟踪\textbf{姿态指令}
    \item 内环控制器计算需要多少\textbf{力矩}来跟踪\textbf{角速率指令}
\end{itemize}

\[
\dot{x}=v,\qquad \dot{v}=a\;(=u),\qquad e:=x_r-x

\textbf{(1) 位置P直控加速度}
\[
a=K_p e
\]
\[
\dot{e}=-v,\qquad \ddot{e}=-\dot{v}=-a=-K_p e
\Rightarrow \boxed{\;\ddot{e}+K_p e=0\;}
\]
\[
s^2+K_p=0 \;\Rightarrow\; s=\pm j\sqrt{K_p}
\]
极点是两个纯虚根，即阻尼比为0，等幅振荡

\medskip
\textbf{(2) 位置外环 + 速度内环（等效PD）}
\[
v_r=K_{p2} e,\qquad a=K_{p1}\,(v_r-v)
   =K_{p1}K_{p2}e-K_{p1}v
\]
\[
\dot{e}=-v,\qquad 
\ddot{e}=-\dot{v}=-a
      =-K_{p1}K_{p2}e+K_{p1}v
      =-K_{p1}K_{p2}e-K_{p1}\dot{e}
\]
\[
\Rightarrow \boxed{\;\ddot{e}+K_{p1}\dot{e}+K_{p1}K_{p2}e=0\;}
\]
与标准二阶系统 \(s^2+2\zeta\omega_n s+\omega_n^2=0\) 对比：
\[
\omega_n=\sqrt{K_{p1}K_{p2}},\qquad
\zeta=\frac{K_{p1}}{2\sqrt{K_{p1}K_{p2}}}
     =\frac{1}{2}\sqrt{\frac{K_{p1}}{K_{p2}}}
\]
\[
\text{过阻尼条件： }\boxed{\;\zeta\ge 1\iff \dfrac{K_{p1}}{K_{p2}}\ge 4\;}
\]
\textit{等效PD： } \(a=k_p e-k_d \dot{x}\) ，其中 \(k_p=K_{p1}K_{p2},\;k_d=K_{p1}\)。
\textit{整定要点：} 令内环快于外环（\(K_{p1}\gg K_{p2}\)）以获得足够阻尼并提前“刹车”。
