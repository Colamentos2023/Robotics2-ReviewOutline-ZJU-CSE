\subsection{定位导航方式}
\begin{enumerate}
    \item 有人工标识导引且路径固定
    \begin{itemize}
        \item \textbf{举例}：磁条导航、磁感应线导航、磁钉导航、二维码导航
        \item \textbf{优点}：技术成熟、稳定可靠、价格优惠
        \item \textbf{缺点}：需要施工和维护，路线无法调整
    \end{itemize}
    \item 有人工标识导引、无固定路径
    \begin{itemize}
        \item \textbf{举例}：激光反射板导引
        \item \textbf{优点}：技术成熟、路径可调
        \item \textbf{缺点}：需要施工和维护、价格昂贵
    \end{itemize}
    \item 无人工标识导引、无固定路径
    \begin{itemize}
        \item \textbf{举例}：自然导航
        \item \textbf{优点}：无需施工、路径可调、连续定位、室内外通用
        \item \textbf{缺点}：算法复杂，环境变化影响定位可靠性和稳定性
    \end{itemize}
\end{enumerate}
